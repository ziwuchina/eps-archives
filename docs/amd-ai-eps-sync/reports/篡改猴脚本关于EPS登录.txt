    // 日志函数，用于显示调试信息
    function log(message, type = 'info') {
        const logDom = document.getElementById('ocr-debug-log');
        if (!logDom) return;
        const time = new Date().toLocaleTimeString();
        const logItem = document.createElement('div');
        logItem.className = `log-${type}`;
        logItem.innerHTML = `[${time}] [${state.step}] ${message}`;
        logDom.appendChild(logItem);
        logDom.scrollTop = logDom.scrollHeight;
        console.log(`[日志] ${message}`);
    }

    // 获取验证码图片元素
    function getCaptchaImage() {
        // 优先通过ID/属性查找验证码图片
        let img = document.querySelector('img[id="IMG1"]') || // 直接ID匹配
                 document.querySelector('img[src*="IdentifyCode.aspx"]') || // 通过图片路径查找
                 document.querySelector('img[alt="验证码"]');

        if (img && img.tagName === 'IMG') {
            log(`找到验证码图片${img.width}x${img.height}，ID/alt:${img.id || img.alt}`, 'info');
            return img;
        }

        // 如果没找到，尝试通过尺寸筛选可能的验证码图片
        const form = document.querySelector('form[action*="login"]') || document.body;
        const images = form.querySelectorAll('img');
        for (const imgEl of images) {
            try {
                if (imgEl.width >= 40 && imgEl.width <= 180 && imgEl.height >= 20 && imgEl.height <= 60) {
                    log(`通过尺寸筛选到可能的验证码图片，src:${imgEl.src.substring(0, 50)}...`, 'info');
                    return imgEl;
                }
            } catch (e) { continue; }
        }

        log(`未找到验证码图片`, 'error');
        return null;
    }

    // 获取验证码输入框元素
    function getCaptchaInput() {
        // 方法1：通过常见的name/id属性查找验证码输入框
        const preciseInput = document.querySelector('input[name="txtCode"]') || // 常见的验证码输入框name
                            document.querySelector('input[id="txtCode"]') ||
                            document.querySelector('input[name="yanzhengma"]') ||
                            document.querySelector('input[id="yanzhengma"]') ||
                            document.querySelector('input[name="validateCode"]') ||
                            document.querySelector('input[id="validateCode"]');

        if (preciseInput) {
            log(`找到验证码输入框，name/id:${preciseInput.name || preciseInput.id}`, 'info');
            return preciseInput;
        }

        // 方法2：通过验证码图片的位置查找相邻的输入框
        const captchaImg = getCaptchaImage();
        if (captchaImg) {
            // 检查验证码图片的兄弟元素
            const siblings = Array.from(captchaImg.parentElement.children);
            for (const sibling of siblings) {
                if (sibling.tagName === 'INPUT' && sibling.type === 'text') {
                    log(`找到验证码图片相邻的输入框`, 'info');
                    return sibling;
                }
            }
            // 检查验证码图片父元素的下一个兄弟元素
            const nextContainer = captchaImg.parentElement.nextElementSibling;
            if (nextContainer) {
                const input = nextContainer.querySelector('input[type="text"]');
                if (input) {
                    log(`找到验证码图片附近的输入框`, 'info');
                    return input;
                }
            }
        }

        // 方法3：通过placeholder/title查找
        const placeholderInput = document.querySelector('input[placeholder="请输入验证码"]') ||
                                document.querySelector('input[placeholder="验证码"]') ||
                                document.querySelector('input[title="验证码"]');

        if (placeholderInput) {
            log(`通过placeholder找到验证码输入框:${placeholderInput.placeholder}`, 'info');
            return placeholderInput;
        }

        // 方法4：排除用户名密码输入框，尝试其他文本框
        const allTextInputs = document.querySelectorAll('input[type="text"]');
        for (const input of allTextInputs) {
            const name = input.name || input.id || '';
            // 排除用户名/账号和密码相关的输入框
            if (!name.includes('user') && !name.includes('name') && !name.includes('zhanghao') &&
                !name.includes('pass') && !name.includes('mima') && !name.includes('password')) {
                log(`找到可能的验证码输入框，name/id:${name}`, 'info');
                return input;
            }
        }

        log(`未找到验证码输入框`, 'error');
        return null;
    }

    // 修复Base64图片格式
    function fixBase64ImageFormat(base64Str) {
        if (!base64Str || typeof base64Str !== 'string') return '';
        if (!base64Str.startsWith('data:image/')) {
            return `data:image/png;base64,${base64Str}`;
        }
        return base64Str;
    }

    // 测试API连接
    function testAPIConnectivity() {
        state.step = 'test-api';
        log('开始测试API连接...', 'step');
        const testBase64 = "data:image/png;base64,iVBORw0KGgoAAAANSUhEUgAAAAEAAAABCAYAAAAfFcSJAAAADUlEQVR42mNk+M9QDwADhgGAWjR9awAAAABJRU5ErkJggg==";

        const requestData = new URLSearchParams();
        requestData.append('apikey', OCR_CONFIG.apiKey);
        requestData.append('language', OCR_CONFIG.language);
        requestData.append('base64Image', testBase64);

        GM_xmlhttpRequest({
            method: 'POST',
            url: OCR_CONFIG.apiUrl,
            headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
            data: requestData.toString(),
            timeout: 10000,
            onload: (res) => {
                if (res.status === 200) {
                    try {
                        const data = JSON.parse(res.responseText);
                        if (data.OCRExitCode === 1) log('API测试成功', 'success');
                        else log(`API错误:${data.ErrorMessage || '未知错误'}`, 'error');
                    } catch (e) { log('API响应解析失败', 'error'); }
                } else {
                    log(`API请求失败，状态码:${res.status}`, 'error');
                }
            },
            onerror: (e) => log(`API请求错误:${e.message}`, 'error'),
            ontimeout: () => log('API请求超时', 'error')
        });
    }

    // 处理OCR请求并填充验证码
    function processOCRRequest(requestData, isJson = false) {
        state.step = 'api-request';
        const headers = isJson ?
            { 'Content-Type': 'application/json' } :
            { 'Content-Type': 'application/x-www-form-urlencoded' };
        const data = isJson ? JSON.stringify(requestData) : requestData.toString();

        GM_xmlhttpRequest({
            method: 'POST',
            url: OCR_CONFIG.apiUrl,
            headers: headers,
            data: data,
            timeout: 15000,
            onload: (res) => {
                if (res.status !== 200) {
                    if (res.status === 405 && !isJson) {
                        log('尝试使用JSON格式请求', 'info');
                        const jsonData = {};
                        requestData.forEach((v, k) => jsonData[k] = v);
                        processOCRRequest(jsonData, true);
                    } else {
                        log(`请求失败，状态码:${res.status}`, 'error');
                    }
                    return;
                }

                try {
                    const data = JSON.parse(res.responseText);
                    if (data.OCRExitCode !== 1) throw new Error(data.ErrorMessage || '识别失败');

                    const result = data.ParsedResults[0]?.ParsedText?.trim() || '';
                    if (result) {
                        log(`识别成功:${result}`, 'success');
                        const input = getCaptchaInput();
                        if (input) {
                            // 设置识别结果到输入框并触发事件
                            input.value = result;
                            input.dispatchEvent(new Event('input', { bubbles: true }));
                            input.dispatchEvent(new Event('change', { bubbles: true }));

                            // 验证设置结果
                            if (input.value === result) {
                                log(`成功将验证码设置到输入框:${input.value}`, 'success');
                            } else {
                                log(`设置验证码值失败，请手动输入`, 'error');
                            }
                        } else {
                            log(`未找到验证码输入框，识别结果:${result}`, 'error');
                        }
                    } else {
                        log('未识别到验证码字符', 'warning');
                    }
                } catch (e) {
                    log(`识别过程中出错:${e.message}`, 'error');
                }
            },
            onerror: (e) => log(`请求错误:${e.message}`, 'error'),
            ontimeout: () => log('识别请求超时', 'error')
        });
    }

    // 开始OCR识别流程
    function startOCR() {
        state.step = 'ocr-start';
        log('开始识别验证码...', 'step');

        const captchaImg = getCaptchaImage();
        const captchaInput = getCaptchaInput();
        if (!captchaImg) { log('未找到验证码图片', 'error'); return; }
        if (!captchaInput) { log('未找到验证码输入框', 'error'); return; }

        try {
            const canvas = document.createElement('canvas');
            canvas.width = captchaImg.width || 100;
            canvas.height = captchaImg.height || 50;
            const ctx = canvas.getContext('2d');
            if (!ctx) throw new Error('无法创建画布上下文');

            try {
                ctx.drawImage(captchaImg, 0, 0);
            } catch (e) {
                log('无法绘制图片，尝试使用URL方式', 'warning');
                const requestData = new URLSearchParams();
                requestData.append('apikey', OCR_CONFIG.apiKey);
                requestData.append('language', OCR_CONFIG.language);
                requestData.append('url', captchaImg.src);
                processOCRRequest(requestData);
                return;
            }

            const base64 = canvas.toDataURL('image/png');
            const fixedBase64 = fixBase64ImageFormat(base64);
            log('成功转换为Base64格式', 'info');

            const requestData = new URLSearchParams();
            requestData.append('apikey', OCR_CONFIG.apiKey);
            requestData.append('language', OCR_CONFIG.language);
            requestData.append('base64Image', fixedBase64);
            processOCRRequest(requestData);

        } catch (e) {
            log(`处理验证码图片失败:${e.message}`, 'error');
        }
    }

    // 创建调试面板

        const panel = document.createElement('div');
        panel.id = 'ocr-debug-panel';

                <button class="ctrl-btn" id="btn-identify">识别</button>
                <button class="ctrl-btn" id="btn-test-api">测试API</button>
            </div>
        `;
        document.body.appendChild(panel);

        // 添加折叠功能
        const toggleBtn = document.getElementById('btn-toggle-panel');
        toggleBtn.addEventListener('click', () => {
            panel.classList.toggle('panel-collapsed');
            toggleBtn.textContent = panel.classList.contains('panel-collapsed') ? '展开' : '收起';
        });

        document.getElementById('btn-reload').addEventListener('click', () => location.reload());
        document.getElementById('btn-identify').addEventListener('click', startOCR);
        document.getElementById('btn-test-api').addEventListener('click', testAPIConnectivity);
    }

    // 初始化自动识别功能
    function initAutoOCR() {
        setTimeout(() => {
            const captchaImg = getCaptchaImage();
            const captchaInput = getCaptchaInput();
            if (captchaImg && captchaInput && !captchaInput.value) {
                log('页面加载完成，开始自动识别', 'info');
                startOCR();
            }

            // 绑定刷新按钮事件
            const refreshBtn = document.querySelector('a:contains("刷新验证码")') ||
                              document.querySelector('span:contains("刷新")') ||
                              document.querySelector('button:contains("刷新")') ||
                              document.querySelector('img[onclick*="refresh"]'); // 查找带refresh事件的图片
            if (refreshBtn) {
                refreshBtn.addEventListener('click', () => {
                    setTimeout(startOCR, 800);
                });
            }
        }, 1500);
    }

    // 主初始化函数
    function init() {
        if (typeof document === 'undefined') return;
        createDebugPanel();
        initAutoOCR();
        log('验证码自动识别脚本初始化完成', 'info');
        log('使用API Key进行验证码识别', 'success');
    }

    init();
})();
