## 4️⃣ Frida Hook 脚本（完整）  

> **文件**: `wrapper/frida/eps_hook.js`  

```js
// ==================== eps_hook.js ====================
if (Process.arch !== 'X86') {
    console.log('[!] 只针对 32‑bit (IA32) 进程。');
    exit();
}

const PROC_NAME = 'eps.exe';
const MODULE    = Process.findModule(PROC_NAME);
if (!MODULE) throw new Error('未找到 eps.exe 对应的模块');

const IMAGE_BASE   = MODULE.base;
const CALLBACK_OF  = 0x433a84;                     // CSSView::CallBackFunction
const CALLBACK_ADDR = IMAGE_BASE.add(CALLBACK_OF);

console.log(`[+] ImageBase = 0x${IMAGE_BASE.toString(16)}`);
console.log(`[+] Hook point = 0x${CALLBACK_ADDR.toString(16)}`);

const LOG_PATH = './logs/eps_callback.log';
function writeLog(msg) {
    const fs = require('fs');
    const LOG_DIR = './logs';
    if (!fs.existsSync(LOG_DIR)) fs.mkdirSync(LOG_DIR);
    const ts = new Date().toISOString();
    const line = `[${ts}] ${msg}\n`;
    fs.appendFileSync(LOG_PATH, line);
    console.log(line.trim());
}

// ------------------- Hook -------------------
Interceptor.attach(CALLBACK_ADDR, {
    onEnter: function (args) {
        try {
            const strPtr = args[1];
            if (strPtr.isNull()) throw new Error('NULL string pointer');
            const MAX_LEN = 1024;
            const raw = Memory.readByteArray(strPtr, MAX_LEN * 2); // Unicode (2 bytes/char)
            const cmd   = raw.toString('utf-16le').replace(/\0/g, '').trim();
            const printable = cmd.replace(/[^\x20-\x7E]/g, '');
            writeLog(`[EPS_HOOK] PID=${Process.getCurrentPid()} \
                     Callback=0x${CALLBACK_ADDR.toString(16)} \
                     Command='${printable}'`);
        } catch (e) {
            writeLog(`[EPS_HOOK][ERROR] ${e.message}`);
        }
    }
});
```

> **使用方式**  
> ```bash
> # 1️⃣ 获取 eps.exe 的 PID（示例 1234）
> frida -p 1234 -l wrapper/frida/eps_hook.js --no-pause
> ```  
> 所有被 `CSSView::CallBackFunction` 传入的命令都会写入 `logs/eps_callback.log`。

## 5️⃣ Python 自动化模板（完整）  

> **文件**: `wrapper/automation.py`  

```python
# ==================== automation.py ====================
import win32gui, win32con, time, json, logging, os
from pathlib import Path

LOG_DIR = Path("./logs")
LOG_DIR.mkdir(exist_ok=True)
logging.basicConfig(filename=LOG_DIR / "eps_automation.log",
                    level=logging.INFO,
                    format="%(asctime)s %(levelname)s %(message)s")

EPS_WINDOW_TITLE = "EPS_main_frame"          # <-- 需要根据实际窗口标题修改
COMMAND_LIST   = ["GET_USER_ROLES", "VERIFY_LICENSE", "SAVE_PROJECT"]

def _hwnd():
    hwnd = win32gui.FindWindowByTitle(EPS_WINDOW_TITLE)
    if not hwnd:
        raise RuntimeError(f"未找到标题为 '{EPS_WINDOW_TITLE}' 的窗口")
    return hwnd

def _send_wm_settext(hwnd, txt: str):
    """向窗口发送 WM_SETTEXT（UTF‑16）"""
    utf16 = txt.encode('utf-16le') + b'\x00\x00'
    win32gui.SendMessage(hwnd, win32con.WM_SETTEXT, 0, utf16)

def _read_window_text(hwnd, max_len=200):
    length = win32gui.GetWindowTextLength(hwnd)
    if length == 0:
        return ""
    buf = win32gui.GetWindowText(hwnd)
    return buf[:max_len]

class EPSAutomation:
    def __init__(self, command_source='commands.json', retry_cfg=None):
        self.hwnd = _hwnd()
        self.command_list = self._load_commands(command_source)
        self.retry_cfg = retry_cfg or {"max_retries": 3, "delay": 0.5}

    @staticmethod
    def _load_commands(p):
        with open(p, 'r', encoding='utf-8') as f:
            data = json.load(f)
        if isinstance(data, list):
            return data
        return data.get("commands", []) or []

    def send_command(self, cmd: str, payload=None, expect_reply=False):
        try:
            full = cmd + (" " + " ".join([f"{k}={v}" for k, v in payload.items()])
                      if payload else cmd)
            _send_wm_settext(self.hwnd, full)
            logging.info(f"[sent] {cmd}")

            if expect_reply:
                time.sleep(0.2)
                reply = _read_window_text(self.hwnd, 200)
                if reply:
                    logging.info(f"[reply] {reply}")
                    if cmd in reply:
                        logging.info("[confirm] reply matches")
                        return 0
                    else:
                        logging.warning("[confirm] reply mismatch")
                        return -1
                else:
                    logging.warning("[confirm] no reply captured")
                    return -1
            return 0
        except Exception as e:
            logging.error(f"[send_error] cmd='{cmd}' - {e}")
            return -1

    def batch_execute(self, expect_reply=False):
        succ = 0
        for cmd in self.command_list:
            attempts = 0
            while attempts < self.retry_cfg["max_retries"]:
                rc = self.send_command(cmd, expect_reply=expect_reply)
                if rc == 0:
                    succ += 1
                    break
                attempts += 1
                time.sleep(self.retry_cfg["delay"])
            else:
                logging.error(f"[failed] cmd='{cmd}' after {self.retry_cfg['max_retries']} retries")
            time.sleep(self.retry_cfg["delay"])
        return succ

    def execute_script(self, path: Path):
        if not path.is_file():
            raise FileNotFoundError(path)
        with path.open('r', encoding='utf-8') as f:
            lines = [ln.strip() for ln in f if ln.strip() and not ln.startswith('#')]
        logging.info(f"[script] executing {len(lines)} commands")
        return self.batch_execute(expect_reply=False)

# ----------------------------------------------------------------------
# 示例入口
# ----------------------------------------------------------------------
if __name__ == "__main__":
    cfg = {"max_retries": 4, "delay_between_retries": 0.6}
    auto = EPSAutomation(command_source="./commands.json", retry_cfg=cfg)
    # 单条示例
    # auto.send_command("GET_USER_ROLES", payload={"mode": "test"}, expect_reply=True)
    # 批量执行
    succ = auto.batch_execute(expect_reply=True)
    logging.info(f"[summary] 成功执行 {succ}/{len(auto.command_list)} 条命令")
```

> **运行示例**  
> ```bash
> python -c "import automation; a = automation.EPSAutomation(); a.batch_execute(expect_reply=True)"
> ```
